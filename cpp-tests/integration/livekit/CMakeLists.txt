cmake_minimum_required(VERSION 3.16)
project(livekit-uniffi-core-probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
if(NOT LIVEKIT_BINDINGS_DIR OR NOT LIVEKIT_FFI_LIBRARY)
    message(FATAL_ERROR "Set LIVEKIT_BINDINGS_DIR and LIVEKIT_FFI_LIBRARY to generated core bindings and their matching Rust library")
endif()
file(GLOB BINDING_SOURCES CONFIGURE_DEPENDS "${LIVEKIT_BINDINGS_DIR}/*.cpp")
add_executable(core-probe core.cpp ${BINDING_SOURCES})
target_include_directories(core-probe PRIVATE "${LIVEKIT_BINDINGS_DIR}")
target_link_libraries(core-probe PRIVATE "${LIVEKIT_FFI_LIBRARY}" Threads::Threads)
if(NOT MSVC)
    target_compile_options(core-probe PRIVATE -pedantic-errors)
endif()
# Keep assert-based checks enabled even if a caller selects a release configuration.
if(MSVC)
    target_compile_options(core-probe PRIVATE /UNDEBUG)
else()
    target_compile_options(core-probe PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME core-probe COMMAND core-probe)
set_tests_properties(core-probe PROPERTIES TIMEOUT 30)
